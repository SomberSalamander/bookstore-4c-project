Table books {
  bookId INT [PK]
  title VARCHAR(50)
  author VARCHAR(50)
  publisher VARCHAR(50)
  description TEXT
  price FLOAT
  stock INT
  img VARCHAR(50)
}

Table users {
  userId INT [PK]
  email VARCHAR(50) [UNIQUE]
  password TEXT
}

Table carts {
  cartId INT [PK]
  userId INT [UNIQUE]
  createdAt DATETIME
}

Table cartItems {
  cartItemId INT [PK]
  cartId INT [note: "FK"]
  bookId INT [note: "FK"]
  quantity INT
}

Ref: carts.cartId < cartItems.cartId

Ref: books.bookId < cartItems.bookId

Ref: carts.userId - users.userId
